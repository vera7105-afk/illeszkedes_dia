    class FiveSlidePresentation {
      constructor() {
        this.currentSlide = 1;
        this.totalSlides = 6;

        this.dom = {
          stageBody: document.getElementById('stageBody'),
          currentSlideText: document.getElementById('currentSlideText'),
          progressBar: document.getElementById('progressBarFill'),
          btnPrev: document.getElementById('btnPrev'),
          btnNext: document.getElementById('btnNext'),
          btnRestart: document.getElementById('btnRestart'),
          btnFullscreen: document.getElementById('btnFullscreen'),
          btnFontSmaller: document.getElementById('btnFontSmaller'),
          btnFontDefault: document.getElementById('btnFontDefault'),
          btnFontLarger: document.getElementById('btnFontLarger'),
          stepDots: document.querySelectorAll('.step-dot'),
          slides: [
            document.getElementById('slide-1'),
            document.getElementById('slide-2'),
            document.getElementById('slide-3'),
            document.getElementById('slide-4'),
            document.getElementById('slide-5'),
            document.getElementById('slide-6')
          ]
        };

        this.init();
      }

      init() {
        this.initFontSize();
        this.bindEvents();
        this.bindMagnetTriggers();
        this.renderSlide(1);
      }

      initFontSize() {
        const saved = localStorage.getItem('illeszkedes_font_size') || 'large';
        this.setFontSize(saved);

        if (this.dom.btnFontSmaller) {
          this.dom.btnFontSmaller.addEventListener('click', () => this.setFontSize('normal'));
        }
        if (this.dom.btnFontDefault) {
          this.dom.btnFontDefault.addEventListener('click', () => this.setFontSize('large'));
        }
        if (this.dom.btnFontLarger) {
          this.dom.btnFontLarger.addEventListener('click', () => this.setFontSize('huge'));
        }
      }

      setFontSize(size) {
        document.documentElement.classList.remove('font-size-normal', 'font-size-large', 'font-size-huge');
        if (this.dom.btnFontSmaller) this.dom.btnFontSmaller.classList.remove('active');
        if (this.dom.btnFontDefault) this.dom.btnFontDefault.classList.remove('active');
        if (this.dom.btnFontLarger) this.dom.btnFontLarger.classList.remove('active');

        if (size === 'normal') {
          document.documentElement.classList.add('font-size-normal');
          if (this.dom.btnFontSmaller) this.dom.btnFontSmaller.classList.add('active');
        } else if (size === 'huge') {
          document.documentElement.classList.add('font-size-huge');
          if (this.dom.btnFontLarger) this.dom.btnFontLarger.classList.add('active');
        } else {
          document.documentElement.classList.add('font-size-large');
          if (this.dom.btnFontDefault) this.dom.btnFontDefault.classList.add('active');
        }
        localStorage.setItem('illeszkedes_font_size', size);
      }

      bindEvents() {
        this.dom.btnNext.addEventListener('click', () => this.nextSlide());
        this.dom.btnPrev.addEventListener('click', () => this.prevSlide());
        this.dom.btnRestart.addEventListener('click', () => this.goToSlide(1));

        this.dom.stepDots.forEach(dot => {
          dot.addEventListener('click', (e) => {
            const slideNum = parseInt(e.currentTarget.getAttribute('data-slide'), 10);
            if (!isNaN(slideNum)) this.goToSlide(slideNum);
          });
        });

        // Billentyűzet kezelés
        window.addEventListener('keydown', (e) => {
          if (e.key === 'ArrowRight' || e.key === 'PageDown' || e.key === ' ') {
            e.preventDefault();
            this.nextSlide();
          } else if (e.key === 'ArrowLeft' || e.key === 'PageUp') {
            e.preventDefault();
            this.prevSlide();
          } else if (e.key === 'Home') {
            e.preventDefault();
            this.goToSlide(1);
          } else if (e.key === 'End') {
            e.preventDefault();
            this.goToSlide(this.totalSlides);
          }
        });

        // Teljes képernyő
        this.dom.btnFullscreen.addEventListener('click', () => {
          if (!document.fullscreenElement) {
            document.documentElement.requestFullscreen().catch(() => {});
          } else {
            if (document.exitFullscreen) document.exitFullscreen();
          }
        });
      }

      bindMagnetTriggers() {
        const btn1 = document.getElementById('btn-replay-1');
        if (btn1) btn1.addEventListener('click', () => this.replayMagnet(1));

        const btn2 = document.getElementById('btn-replay-2');
        if (btn2) btn2.addEventListener('click', () => this.replayMagnet(2));

        const btn3 = document.getElementById('btn-replay-3');
        if (btn3) btn3.addEventListener('click', () => this.replayMagnet(3));
      }

      replayMagnet(slideNum) {
        const attract = document.getElementById(`chip-attract-${slideNum}`);
        const repel = document.getElementById(`chip-repel-${slideNum}`);
        if (attract && repel) {
          attract.classList.remove('chip-attracted');
          repel.classList.remove('chip-repelled');
          void attract.offsetWidth;
          void repel.offsetWidth;
          attract.classList.add('chip-attracted');
          repel.classList.add('chip-repelled');
        }
      }

      nextSlide() {
        if (this.currentSlide < this.totalSlides) {
          this.goToSlide(this.currentSlide + 1);
        }
      }

      prevSlide() {
        if (this.currentSlide > 1) {
          this.goToSlide(this.currentSlide - 1);
        }
      }

      goToSlide(slideNum) {
        if (slideNum < 1 || slideNum > this.totalSlides) return;
        this.currentSlide = slideNum;
        this.renderSlide(this.currentSlide);
      }

      renderSlide(slideNum) {
        // Gördítés visszaállítása a tetejére minden diaváltásnál
        if (this.dom.stageBody) {
          this.dom.stageBody.scrollTop = 0;
        }

        this.dom.currentSlideText.textContent = slideNum;
        this.dom.progressBar.style.width = `${(slideNum / this.totalSlides) * 100}%`;

        this.dom.btnPrev.disabled = (slideNum === 1);
        this.dom.btnNext.disabled = (slideNum === this.totalSlides);

        this.dom.stepDots.forEach(dot => {
          const num = parseInt(dot.getAttribute('data-slide'), 10);
          dot.classList.remove('active', 'passed');
          if (num === slideNum) dot.classList.add('active');
          else if (num < slideNum) dot.classList.add('passed');
        });

        this.dom.slides.forEach((slide, idx) => {
          if (idx + 1 === slideNum) {
            slide.classList.add('active');
          } else {
            slide.classList.remove('active');
          }
        });

        if (slideNum >= 1 && slideNum <= 3) {
          this.replayMagnet(slideNum);
        }
      }
    }

    class IlleszkedesGame {
      constructor() {
        this.questions = [];
        this.currentIdx = 0;
        this.score = 0;
        this.total = GAME_QUESTIONS.length;
        this.locked = false;

        this.dom = {
          counter: document.getElementById('gameQuestionCounter'),
          scoreBadge: document.getElementById('gameScoreBadge'),
          stemWord: document.getElementById('gameStemWord'),
          stemTag: document.getElementById('gameStemTag'),
          questionText: document.getElementById('gameQuestionText'),
          optionsContainer: document.getElementById('gameOptionsContainer'),
          feedbackBox: document.getElementById('gameFeedbackBox'),
          playCard: document.getElementById('gamePlayCard'),
          finishedCard: document.getElementById('gameFinishedCard'),
          finalScore: document.getElementById('gameFinalScore'),
          btnRestart: document.getElementById('btnRestartGame')
        };

        this.init();
      }

      init() {
        if (this.dom.btnRestart) {
          this.dom.btnRestart.addEventListener('click', () => this.restart());
        }
        this.restart();
      }

      shuffleQuestions() {
        this.questions = JSON.parse(JSON.stringify(GAME_QUESTIONS));
        for (let i = this.questions.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [this.questions[i], this.questions[j]] = [this.questions[j], this.questions[i]];
        }
      }

      restart() {
        this.shuffleQuestions();
        this.currentIdx = 0;
        this.score = 0;
        this.locked = false;
        this.dom.playCard.style.display = 'flex';
        this.dom.finishedCard.style.display = 'none';
        this.renderQuestion();
      }

      renderQuestion() {
        if (this.currentIdx >= this.total) {
          this.showFinished();
          return;
        }

        const q = this.questions[this.currentIdx];
        this.locked = false;

        this.dom.counter.textContent = `Kérdés: ${this.currentIdx + 1} / ${this.total}`;
        this.dom.scoreBadge.textContent = `Pontszám: ${this.score} pont`;
        this.dom.stemWord.textContent = q.stem;
        this.dom.stemTag.textContent = `${this.currentIdx + 1}. tanuló • ${q.tag}`;
        this.dom.questionText.textContent = q.prompt;

        this.dom.feedbackBox.className = 'game-feedback-box';
        this.dom.feedbackBox.textContent = '';

        const shuffledOptions = q.options.map((optText, origIdx) => ({
          text: optText,
          isCorrect: (origIdx === q.correct)
        }));

        for (let i = shuffledOptions.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [shuffledOptions[i], shuffledOptions[j]] = [shuffledOptions[j], shuffledOptions[i]];
        }

        this.dom.optionsContainer.innerHTML = '';
        shuffledOptions.forEach((item) => {
          const btn = document.createElement('button');
          btn.className = 'btn-game-opt';
          btn.type = 'button';
          btn.textContent = item.text;
          btn.addEventListener('click', () => this.handleAnswer(item.isCorrect, btn));
          this.dom.optionsContainer.appendChild(btn);
        });
      }

      handleAnswer(isCorrect, clickedBtn) {
        if (this.locked) return;
        this.locked = true;

        const q = this.questions[this.currentIdx];
        const allButtons = this.dom.optionsContainer.querySelectorAll('.btn-game-opt');

        if (isCorrect) {
          this.score++;
          this.dom.scoreBadge.textContent = `Pontszám: ${this.score} pont`;
          clickedBtn.classList.add('opt-correct');
          this.dom.feedbackBox.className = 'game-feedback-box feedback-success show';
          this.dom.feedbackBox.innerHTML = `🎉 ${q.explanation}`;
        } else {
          clickedBtn.classList.add('opt-wrong');
          allButtons.forEach(btn => {
            if (btn.textContent === q.options[q.correct]) {
              btn.classList.add('opt-correct');
            }
          });
          this.dom.feedbackBox.className = 'game-feedback-box feedback-error show';
          this.dom.feedbackBox.innerHTML = `❌ ${q.explanation}`;
        }

        setTimeout(() => {
          this.currentIdx++;
          this.renderQuestion();
        }, 2200);
      }

      showFinished() {
        this.dom.playCard.style.display = 'none';
        this.dom.finishedCard.style.display = 'flex';
        this.dom.finalScore.textContent = `Végeredmény: ${this.score} / ${this.total} pont (${Math.round(this.score / this.total * 100)}%)`;
      }
    }

    document.addEventListener('DOMContentLoaded', () => {
      window.presentation = new FiveSlidePresentation();
      window.game = new IlleszkedesGame();
    });
  </script>
